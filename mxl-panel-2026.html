async function publicarDirecto() { const status = document.getElementById('status-msg'); status.style.display = 'block'; status.innerText = "⏳ Conectando con Railway..."; const producto = { tEn: document.getElementById('tEn').value, tEs: document.getElementById('tEs').value, dEn: document.getElementById('dEn').value, dEs: document.getElementById('dEs').value, img: document.getElementById('img').value, link: document.getElementById('link').value, fecha: new Date().toISOString() }; // Validar campos obligatorios if (!producto.tEn || !producto.link) { status.innerText = "❌ Error: Título y Link son obligatorios"; status.style.background = "rgba(255,69,0,0.2)"; status.style.color = "#ff4500"; return; } try { const res = await fetch('/api/publicar', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(producto) }); const data = await res.json(); if(data.success) { status.className = "success"; status.style.background = "rgba(40,167,69,0.2)"; status.style.color = "#2ecc71"; status.innerText = "✅ ¡PUBLICADO! El producto ya está en vivo en Ideal Home."; // Limpiar campos después de 2 segundos setTimeout(() => { document.querySelectorAll('#tab-publisher input, #tab-publisher textarea').forEach(i => i.value = ''); status.style.display = 'none'; }, 2000); } else { status.innerText = "❌ Error: " + (data.error || "No se pudo publicar"); } } catch (e) { console.error(e); status.innerText = "❌ Error: No se pudo conectar con el servidor"; status.style.background = "rgba(255,69,0,0.2)"; status.style.color = "#ff4500"; } }
