const express = require('express'); const path = require('path'); const fs = require('fs'); const app = express(); const PORT = process.env.PORT || 8080; app.use(express.json()); app.use(express.static('.')); // Sirve tus archivos HTML const DB_PATH = path.join(__dirname, 'productos.json'); // Si no existe la base de datos, la creamos if (!fs.existsSync(DB_PATH)) { fs.writeFileSync(DB_PATH, JSON.stringify([])); } // 🚀 ESTA ES LA RUTA QUE TE FALTA (La que quita el error 404) app.post('/api/publicar', (req, res) => { try { const data = JSON.parse(fs.readFileSync(DB_PATH)); data.unshift(req.body); // Guarda el producto nuevo al principio fs.writeFileSync(DB_PATH, JSON.stringify(data, null, 2)); res.json({ success: true }); } catch (error) { res.status(500).json({ success: false, error: error.message }); } }); // Ruta para que el index lea los productos app.get('/api/productos', (req, res) => { const data = JSON.parse(fs.readFileSync(DB_PATH)); res.json(data); }); app.listen(PORT, () => { console.log(`🚀 MXL GOLD MINER activo en puerto ${PORT}`); });
